# G-05 CAS 单点登录（后端闭环 + 桌面端接入）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 synapse-rust 的 CAS 登录链路真正可用（内置 CAS 登录页 → 签发 ST → Matrix SSO 回调换 loginToken），并让 TJG 桌面端通过 Tauri 子窗口拦截 loginToken 完成登录。

**Architecture:** synapse-rust 作为自包含 CAS Server：Matrix 客户端入口 302 到 `/_synapse/cas/login?service=<callback>`，登录页用矩阵账号密码认证并创建 service ticket，重定向回 callback；callback 本地校验 ticket 后签发 loginToken，重定向回 `redirect_after?loginToken=`。TJG 侧用子 WebviewWindow 打开 SSO 入口，Rust `on_navigation` 拦截最终 URL 提取 token 并广播事件，前端复用会话层完成登录。

**Tech Stack:** Rust / axum（synapse-web）、CAS 存储（`cas_tickets`）、`LoginTokenService`；Vue 3 + Pinia + Tauri v2（`WebviewWindowBuilder::on_navigation`）；Vitest。

```rust
/// Matrix 客户端 CAS 回调路径，`service` 只允许指向它，避免登录页被当成开放跳板。
const MATRIX_CAS_CALLBACK_PATH: &str = "/_matrix/client/v3/login/sso/redirect/cas/callback";

fn matrix_cas_service(redirect_after: &str) -> String {
    format!(
        "{MATRIX_CAS_CALLBACK_PATH}?redirect_after={}",
        urlencoding::encode(redirect_after)
    )
}

fn is_matrix_cas_service(service: &str) -> bool {
    let prefix = format!("{MATRIX_CAS_CALLBACK_PATH}?redirect_after=");
    service.starts_with(&prefix)
        && service.len() > prefix.len()
        && !service
            .bytes()
            .any(|b| b < 0x20 || b == 0x7f || b == b'\\')
}

fn append_ticket(service: &str, ticket: &str) -> String {
    let sep = if service.contains('?') { '&' } else { '?' };
    format!("{service}{sep}ticket={}", urlencoding::encode(ticket))
}

fn build_client_redirect(redirect_after: &str, login_token: &str) -> String {
    let sep = if redirect_after.contains('?') { '&' } else { '?' };
    format!("{redirect_after}{sep}loginToken={}", urlencoding::encode(login_token))
}

fn escape_html(value: &str) -> String {
    value
        .replace('&', "&amp;")
        .replace('<', "&lt;")
        .replace('>', "&gt;")
        .replace('"', "&quot;")
        .replace('\'', "&#39;")
}
```

```rust
async fn cas_login_page(
    State(_ctx): State<SsoContext>,
    MatrixQuery(query): MatrixQuery<ServiceTicketQuery>,
) -> Result<Response, ApiError> {
    if !is_matrix_cas_service(&query.service) {
        return Err(ApiError::invalid_param(
            "service must be the Matrix CAS callback path",
        ));
    }
    Ok(html_response(StatusCode::OK, render_login_page(&query.service, None)))
}

async fn cas_login_submit(
    State(ctx): State<SsoContext>,
    Form(form): Form<CasLoginForm>,
) -> Result<Response, ApiError> {
    if !is_matrix_cas_service(&form.service) {
        return Err(ApiError::invalid_param(
            "service must be the Matrix CAS callback path",
        ));
    }
    if form.username.trim().is_empty() || form.password.is_empty() {
        return Ok(html_response(
            StatusCode::UNAUTHORIZED,
            render_login_page(&form.service, Some("请输入用户名和密码")),
        ));
    }

    let user = match ctx
        .account_identity_service
        .get_user_by_identifier(form.username.trim())
        .await?
    {
        Some(user) => user,
        None => {
            return Ok(html_response(
                StatusCode::UNAUTHORIZED,
                render_login_page(&form.service, Some("用户名或密码错误")),
            ))
        }
    };
    let user_id = user.user_id().to_string();

    if let Err(err) = ctx
        .credential_auth
        .verify_user_credentials(&user_id, &form.password)
        .await
    {
        if err.is_forbidden() || err.is_unauthorized() || err.is_bad_request() {
            return Ok(html_response(
                StatusCode::UNAUTHORIZED,
                render_login_page(&form.service, Some("用户名或密码错误")),
            ));
        }
        return Err(err);
    }

    let ticket = ctx.cas_service.create_service_ticket(&user_id, &form.service).await?;
    let location = append_ticket(&form.service, &ticket.ticket_id);
    Ok((StatusCode::FOUND, [(header::LOCATION, location)]).into_response())
}
```

```rust
#[derive(Debug, Deserialize)]
struct SsoCallbackQuery {
    redirect_after: Option<String>,
    ticket: Option<String>,
}

async fn cas_sso_callback(
    State(ctx): State<SsoContext>,
    MatrixQuery(query): MatrixQuery<SsoCallbackQuery>,
) -> Result<Response, ApiError> {
    let redirect_after = query
        .redirect_after
        .filter(|value| !value.is_empty())
        .ok_or_else(|| ApiError::missing_param("Missing required query parameter: redirect_after"))?;
    if !is_same_origin_path(&redirect_after) {
        return Err(ApiError::invalid_param("redirect_after must be a same-origin path"));
    }
    let ticket = query
        .ticket
        .filter(|value| !value.is_empty())
        .ok_or_else(|| ApiError::missing_param("Missing required query parameter: ticket"))?;

    let service = matrix_cas_service(&redirect_after);
    let validated = ctx
        .cas_service
        .validate_service_ticket(&ticket, &service)
        .await?
        .ok_or_else(|| ApiError::forbidden("Invalid or expired CAS ticket"))?;

    let login_token = ctx
        .login_token_service
        .generate(&validated.user_id, None)
        .await?;

    Ok((
        StatusCode::FOUND,
        [(header::LOCATION, build_client_redirect(&redirect_after, &login_token))],
    )
        .into_response())
}
```

```rust
use tauri::{AppHandle, Emitter, Manager, Url, WebviewUrl, WebviewWindowBuilder};

const CAS_WINDOW_LABEL: &str = "cas-login";
pub const CAS_LOGIN_TOKEN_EVENT: &str = "cas-login-token";

fn extract_login_token(url: &str) -> Option<String> {
    let parsed = Url::parse(url).ok()?;
    parsed
        .query_pairs()
        .find(|(key, _)| key == "loginToken" || key == "login_token")
        .map(|(_, value)| value.into_owned())
        .filter(|value| !value.is_empty())
}

#[tauri::command]
pub async fn open_cas_login_window(
    app: AppHandle,
    login_url: String,
    homeserver_origin: String,
) -> Result<(), String> {
    let login = Url::parse(&login_url).map_err(|e| e.to_string())?;
    let homeserver = Url::parse(&homeserver_origin).map_err(|e| e.to_string())?;
    if login.scheme() != "http" && login.scheme() != "https" {
        return Err("CAS 登录地址必须使用 http/https".to_string());
    }
    if login.origin() != homeserver.origin() {
        return Err("CAS 登录地址必须与家服务器同源".to_string());
    }

    if let Some(existing) = app.get_webview_window(CAS_WINDOW_LABEL) {
        let _ = existing.close();
    }

    let handle = app.clone();
    WebviewWindowBuilder::new(&app, CAS_WINDOW_LABEL, WebviewUrl::External(login))
        .title("CAS 登录")
        .inner_size(480.0, 640.0)
        .resizable(false)
        .on_navigation(move |url| {
            if let Some(token) = extract_login_token(url.as_str()) {
                let _ = handle.emit(CAS_LOGIN_TOKEN_EVENT, token);
                if let Some(window) = handle.get_webview_window(CAS_WINDOW_LABEL) {
                    let _ = window.close();
                }
                return false;
            }
            true
        })
        .build()
        .map_err(|e| e.to_string())?;
    Ok(())
}
```

```ts
import { invoke } from '@tauri-apps/api/core'
import { listen, type UnlistenFn } from '@tauri-apps/api/event'
import { onBeforeUnmount, ref } from 'vue'
import { MatrixAuthService } from '@/services/matrix/auth/MatrixAuthService'
import { resolveMatrixRuntimeEndpointConfig } from '@/services/backend'

const CAS_LOGIN_TOKEN_EVENT = 'cas-login-token'

export function useCasLogin(options: {
  onToken: (loginToken: string) => Promise<void>
}) {
  const casAvailable = ref(false)
  let unlisten: UnlistenFn | undefined

  const detect = async () => {
    casAvailable.value = await MatrixAuthService.hasCasLoginFlow()
  }

  const registerListener = async () => {
    unlisten = await listen<string>(CAS_LOGIN_TOKEN_EVENT, (event) => {
      void options.onToken(event.payload)
    })
  }

  const startCasLogin = async () => {
    if (!casAvailable.value) return
    const loginUrl = await MatrixAuthService.getCasLoginUrl('/')
    const homeserverOrigin = resolveMatrixRuntimeEndpointConfig().homeserverUrl
    await invoke('open_cas_login_window', { loginUrl, homeserverOrigin })
  }

  onBeforeUnmount(() => {
    unlisten?.()
  })

  return { casAvailable, detect, registerListener, startCasLogin }
}
```
